```ts
// engine/src/schema/node-table.ts —— T1 产出
export type NodeKind = 'action' | 'gate' | 'terminal'
export interface FlowNode { id: string; name?: string; kind: NodeKind; stage?: string;
  emp?: string; prompt?: string; model_tier?: string; human_gate?: boolean;
  gate?: string; next: string[] }
export interface GateSpec { kind: 'review' | 'acceptance' | 'decision';
  reviewer: string; on_pass?: string; on_reflow?: string; branches?: Record<string, string>;
  covers?: string[] }
export interface NodeTable { flow: string; display_name?: string; version?: number;
  max_retries: number; terminal_fail: string | null; delivery_node?: string;
  nodes: FlowNode[]; gate_specs: Record<string, GateSpec> }
export type TableError = { ok: false; error: string; detail?: string }
export function parseNodeTable(raw: unknown): NodeTable          // zod+跨字段校验，非法抛 EngineSchemaError（含字段定位）
export interface LoadedTable extends NodeTable { nodeMap: Map<string, FlowNode>;
  adjacency: Map<string, string[]> }                             // T2 产出：归一化视图
export function loadNodeTable(table: NodeTable): LoadedTable

// engine/src/schema/events.ts —— T1 产出
export type EventType = 'run.created'|'run.completed'|'run.aborted'|'transition'|'gate'|'dispatch'
export interface EngineEvent { seq: number; ts: string; type: EventType; trace_id: string;
  parent_seq: number | null; actor: string; flow: string;
  // 各类载荷（可选键按 type 出现）：
  from?: string; to?: string; reflow?: boolean; forced_fail?: boolean; reason?: string | null;
  status?: string; gate?: string; kind?: string; node?: string; verdict?: string;
  iter?: number; reviewer?: string; issues?: string[]; request_id?: string;
  phase?: 'start'|'done'; emp?: string; dispatch_id?: string; usage?: Record<string, number>;
  title?: string; workspace?: string; display_name?: string; final_node?: string; duration_s?: number }

// engine/src/engine.ts —— T5 产出（11 操作，§9.1；zod 入参）
export interface CreateTaskInput { mode: 'team'|'solo'; flow?: string; employee?: string;
  workspace: string; title: string; input: string; base?: string; model?: string; effort?: string }
export interface EngineDeps { dataDir: string; templatesDir: string }   // 注入：~/.devzero 与 templates/flows
export class Engine {
  constructor(deps: EngineDeps)
  onEvent(listener: (e: EngineEvent) => void): () => void   // SSE/驱动器订阅源
  createTask(input: CreateTaskInput): { task_id: string }
  getTask(taskId: string): TaskView
  nextStep(taskId: string): NextStepView
  advance(taskId: string, to: string, opts?: { reflow?: boolean; reason?: string; actor?: string }): TaskView
  recordGate(taskId: string, g: { gate: string; verdict: 'PASS'|'FAIL'|string; by: string;
    issues?: string[]; request_id?: string }): TaskView
  dispatchStart(taskId: string, p: { emp: string; node?: string; prompt?: string }): { dispatch_id: string }
  dispatchDone(taskId: string, p: { emp: string; dispatch_id: string; status?: 'done'|'blocked';
    usage?: Record<string, number> }): TaskView
  handoffWrite(taskId: string, p: { emp: string; node: string; summary: string;
    artifacts?: string[]; status?: 'done'|'blocked'|'needs_context'; reason?: string }): { path: string }
  confirmGate(taskId: string, p: { node: string; verdict: 'approve'|'reject'; note?: string }): TaskView
  completeTask(taskId: string, status?: 'completed'|'aborted'): TaskView
  getTable(taskId: string): NodeTable
  readEvents(taskId: string, afterSeq?: number): EngineEvent[]          // SSE 重放/HTTP 拉取共用
}
export interface TaskView { task_id: string; flow: string; title: string; workspace: string;
  status: 'in_progress'|'gate_paused'|'blocked'|'completed'|'aborted';
  current_node: string | null; gate_iters: Record<string, number>; gate_calls: number;
  retries: Record<string, number>; blocked_reason: string | null; position?: { cleared: number; total: number; pct: number } }
export interface NextStepView { current_node: string | null; node_kind: NodeKind | null;
  node_name: string | null; emp?: string; prompt?: string; next_actions: { to_node: string; label: string }[];
  gate_spec?: GateSpec & { gate: string; current_iter: number; max_retries: number };
  is_blocked: boolean; blocked_reason: string | null }
```
